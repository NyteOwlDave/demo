function pcl( o ) {
    o = pcl.prep( o );
    if (! Array.isArray( o ) ) {
       o = o.split( "\n" );
    }
    return ( 
        ( o )
        . map( str )
        . filter( s => s )
    );
}

pcl.prep = function( o ) {
    if ( Array.isArray( o ) ) {
        return ( o );
    }
    if ( o instanceof Set ) {
        return arr( o );
    }
    if ( o instanceof Object ) {
        return mem( o );
    }
    if ( "function" === typeof o ) {
        return mem( o );
    }
    return String( o );
};

function assist( o, title ) {
    const t = ( str( title ) || "Members" );
    o = pcl( o );
    o . unshift( `[ ${t} ]\n` );
    blurt( ( o ).join( "\n" ) );
}
