function visit( url ) {
    if ( null === localStorage ) {
        const a = elx( "A" );
        a . href = ( url );
        a . click();
    } else {
        const w = window;
        w.open( url, url );
    }
}
</script>

<script>
visit.demo = function( event ) {
    try {
        let sender = event.target;
        if (! sender.classList.contains( "cool-icon" ) ) {
            sender = sender.parentElement;
        }
        const url = sender.getAttribute( "url" );
        visit( url );
    } catch ( e ) {
        alert ( e )
        throw ( e )
    }
};
</script>
