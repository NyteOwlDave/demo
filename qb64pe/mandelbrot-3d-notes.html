Rotation Matrix


function mtx_rotate( x, y, z, m ) {

    c = cos( theta );
    s = sin( theta );
    t = 1 - c;

    n = normal_vec( x, y, z )
    x = n[ 0 ]
    y = n[ 1 ]
    z = n[ 2 ]

    m[0][0] = t*x*x + c;
    m[0][1] = t*x*y - z*s;
    m[0][2] = t*x*z + y*s;

    m[1][0] = t*x*y + z*s;
    m[1][1] = t*y*y + c;
    m[1][2] = t*y*z - x*s;

    m[2][0] = t*x*z - y*s;
    m[2][1] = t*y*z + x*s;
    m[2][2] = t*z*z + c;

}
